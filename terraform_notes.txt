#Deploy Databricks jobs: 
# One-time: cd into the shared terraform working directory (same for every job)
cd "$CENTRAL_REPO/terraform/databricks/objects/job"

# 1. Point the backend at job-a's own remote state file, re-initializing providers/backend
terraform init -input=false -reconfigure \
  -backend-config="address=https://jfrog.example.com/terraform-state/databricks/dev/job/job-a.tfstate" \
  -backend-config="update_method=PUT"

# 2. Apply using job-a's tfvars file (lives in a DIFFERENT directory, referenced by full path)
terraform apply -input=false -auto-approve \
  -var environment=dev \
  -var-file="$GITHUB_WORKSPACE/job-tfvars/job-a.tfvars.json"

# 3. Re-point the backend at job-b's own remote state file
terraform init -input=false -reconfigure \
  -backend-config="address=https://jfrog.example.com/terraform-state/databricks/dev/job/job-b.tfstate" \
  -backend-config="update_method=PUT"

# 4. Apply using job-b's tfvars file
terraform apply -input=false -auto-approve \
  -var environment=dev \
  -var-file="$GITHUB_WORKSPACE/job-tfvars/job-b.tfvars.json"

#Deploy Databricks Sql Warehouse: 
cd /terraform/databricks/objects/sql-warehouse
terraform apply -input=false -auto-approve \
  -var environment=dev \
  -var sql_warehouse_name=wh-a \
  -var sql_warehouse_sub_app_name=<SubAppName> \
  -var sql_warehouse_app_master=<AppMaster> \
  -var sql_warehouse_billing_cluster_name=<BillingClusterName> \
  -var sql_warehouse_cluster_size=Medium \
  -var sql_warehouse_auto_stop_mins=10

extra tags are also considered when tags are in tags.auto.tfvars.json 
For wh-a with extra tags `CostCenter` and `Owner`:

```json
{
  "tags": {
    "CostCenter": "CC1234",
    "Owner": "jane.doe"
  }
}
```
